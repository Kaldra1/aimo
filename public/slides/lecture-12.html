Сортування Шелла: ідея

Проблема вставок

Елемент рухається до свого місця лише на одну позицію за раз.

void shellSort(vector<int>& a) {
    int n = a.size();
    for (int h = n / 2; h > 0; h /= 2) {
        for (int i = h; i < n; i++) {  // вставки з кроком h
            int key = a[i];
            int j = i;
            while (j >= h && a[j - h] > key) {
                a[j] = a[j - h];
                j -= h;
            }
            a[j] = key;
        }
    }
}

Злиття: реалізація й аналіз

C++
void mergeParts(vector<int>& a, int left, int mid, int right,
                vector<int>& buffer) {
    int i = left, j = mid, k = left;
    while (i < mid && j < right)
        buffer[k++] = (a[i] <= a[j]) ? a[i++] : a[j++];
    while (i < mid) buffer[k++] = a[i++];
    while (j < right) buffer[k++] = a[j++];
    for (int t = left; t < right; t++) a[t] = buffer[t];
}

void mergeSort(vector<int>& a, int left, int right, vector<int>& buffer) {
    if (right - left <= 1) return;   // базовий випадок
    int mid = left + (right - left) / 2;
    mergeSort(a, left, mid, buffer);
    mergeSort(a, mid, right, buffer);
    mergeParts(a, left, mid, right, buffer);
}

void mergeSort(vector<int>& a) {
    vector<int> buffer(a.size());
    mergeSort(a, 0, a.size(), buffer);
}

Швидке сортування: ідея

Ч. Хоар, 1960
1Вибрати опорний елемент (pivot)
2Розбити: ліворуч менші, праворуч не менші, опорний — на остаточному місці
3Рекурсивно відсортувати ліву й праву частини
Робота — до рекурсивних викликів; «зливати» потім нічого не треба
Розбиття Ломуто

Опорний — останній елемент. Індекс i вказує на місце для наступного «малого». Ідемо індексом j: якщо aⱼ < pivot — обмін aᵢ ↔ aⱼ, i++. Наприкінці опорний — на позицію i.

Швидке: реалізація й аналіз

C++
int partition(vector<int>& a, int low, int high) {
    int pivot = a[high];
    int i = low;
    for (int j = low; j < high; j++) {
        if (a[j] < pivot) {
            swap(a[i], a[j]);
            i++;
        }
    }
    swap(a[i], a[high]);  // опорний на місце
    return i;
}

void quickSort(vector<int>& a, int low, int high) {
    if (low >= high) return;
    int p = partition(a, low, high);
    quickSort(a, low, p - 1);
    quickSort(a, p + 1, high);
}

Середній випадок

O(n log n); зазвичай найшвидший серед порівняльних

Найгірший випадок

O(n²) — на вже відсортованому масиві з опорним «останній»

Як уникнути

Випадковий опорний або медіана трьох (першого, середнього, останнього)

На місці, нестійкий

Пам'ять — стек рекурсії, у середньому O(log n)

Пірамідальне: реалізація й аналіз

C++
void siftDown(vector<int>& a, int i, int size) {
    while (true) {
        int largest = i;
        int left = 2 * i + 1, right = 2 * i + 2;
        if (left < size && a[left] > a[largest]) largest = left;
        if (right < size && a[right] > a[largest]) largest = right;
        if (largest == i) return;
        swap(a[i], a[largest]);
        i = largest;
    }
}

void heapSort(vector<int>& a) {
    int n = a.size();
    for (int i = n / 2 - 1; i >= 0; i--) siftDown(a, i, n);
    for (int end = n - 1; end > 0; end--) {
        swap(a[0], a[end]);  // максимум у кінець
        siftDown(a, 0, end);
    }
}

Просіювання

Не довше висоти дерева — log₂n рівнів

Етапи

Побудова купи O(n), вибір максимумів O(n log n)

Гарантія

O(n log n) навіть у найгіршому випадку, без додаткового масиву

Нестійкий

Трохи повільніший за швидке через «стрибки» пам'яттю

void countingSort(vector<int>& a, int k) {
    vector<int> count(k + 1, 0);
    for (int x : a) count[x]++;
    int pos = 0;
    for (int value = 0; value <= k; value++)
        while (count[value]-- > 0) a[pos++] = value;
}

O(n + k) часу, O(k) пам'яті

Оцінки, вік, байти (k = 255) — найшвидший метод

Обмеження

Великий k — не підходить. Стійка версія — через префіксні суми, основа radix sort

Що забрати з лекції

01Ефективні методи мають трудомісткість O(n log n) замість O(n²)
02Шелл упорядковує далекі елементи, потім зменшує крок; на місці, нестійкий
03Злиття: Θ(n log n) завжди, стійке, але O(n) додаткової пам'яті
04Швидке в середньому найшвидше; поганий опорний — O(n²), рятують випадковий опорний чи медіана трьох
05Пірамідальне: O(n log n) у найгіршому випадку без додаткової пам'яті
06Підрахунок не порівнює елементи: O(n + k) для цілих з малого діапазону
