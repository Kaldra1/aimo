Базові приклади

Сума цифр

int digitSum(int n) {
  if (n < 10) return n;
  return n % 10
       + digitSum(n / 10);
}
4096 → 6 + 409
→ 6 + 9 + 40
→ 6 + 9 + 0 + 4 = 19

НСД · Евклід

int gcd(int a, int b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
gcd(48, 18) → (18, 12)
→ (12, 6) → (6, 0)
= 6

bool isPal(const string& s,
           int l, int r) {
  if (l >= r) return true;
  if (s[l] != s[r]) return false;
  return isPal(s, l + 1, r - 1);
}

double power(double a, int n) {
    if (n == 0) return 1;
    if (n % 2 == 0) {
        double half = power(a, n / 2);
        return half * half;
    }
    return a * power(a, n - 1);
}

long long fibMemo(int n,
        vector<long long>& memo) {
  if (n <= 1) return n;
  if (memo[n] != -1)
      return memo[n];
  return memo[n] =
      fibMemo(n-1, memo)
    + fibMemo(n-2, memo);
}

Ітерація

O(n)
long long fibIter(int n) {
  long long prev = 0, cur = 1;
  for (int i = 0; i < n; i++) {
    long long next = prev + cur;
    prev = cur;
    cur = next;
  }
  return prev;
}
